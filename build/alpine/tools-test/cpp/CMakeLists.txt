cmake_minimum_required(VERSION 3.20)
project(toolsprobe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(benchmark REQUIRED)
find_package(Threads REQUIRED)
add_executable(cpp-hello hello.cpp)
target_compile_options(cpp-hello PRIVATE -Wall -Wextra -Werror)
target_link_libraries(cpp-hello PRIVATE Threads::Threads)
add_executable(cpp-bench bench.cpp)
target_link_libraries(cpp-bench PRIVATE benchmark::benchmark_main Threads::Threads)
enable_testing()
add_test(NAME hello COMMAND cpp-hello)
